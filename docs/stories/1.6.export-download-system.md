### PDF Generation Template
```python
from reportlab.lib.pagesizes import letter
from reportlab.platypus import SimpleDocTemplate, Paragraph, PageBreak
from reportlab.lib.styles import getSampleStyleSheet, ParagraphStyle
from reportlab.pdfbase import pdfmetrics
from reportlab.pdfbase.ttfonts import TTFont

class PDFExporter:
    def generate_pdf(self, document: Document, translation: Translation) -> bytes:
        # Register fonts for different languages
        pdfmetrics.registerFont(TTFont('Devanagari', 'NotoSansDevanagari.ttf'))
        
        # Create PDF with metadata
        buffer = BytesIO()
        pdf = SimpleDocTemplate(buffer, pagesize=letter)
        
        # Build content with structure preservation
        story = []
        styles = getSampleStyleSheet()
        
        # Add translated content with formatting
        for paragraph in translation.paragraphs:
            style = self.get_paragraph_style(paragraph)
            story.append(Paragraph(paragraph.text, style))
            
        pdf.build(story)
        return buffer.getvalue()
```

### DOCX Export Implementation
```python
from docx import Document
from docx.shared import Pt, RGBColor
from docx.enum.text import WD_ALIGN_PARAGRAPH

class DOCXExporter:
    def generate_docx(self, document_data: dict) -> bytes:
        doc = Document()
        
        # Set document properties
        doc.core_properties.title = document_data['title']
        doc.core_properties.subject = f"Translation: {document_data['source_lang']} to {document_data['target_lang']}"
        doc.core_properties.creator = "Translation Platform"
        
        # Add content with formatting
        for paragraph_data in document_data['paragraphs']:
            p = doc.add_paragraph()
            p.alignment = self.get_alignment(paragraph_data)
            run = p.add_run(paragraph_data['text'])
            run.font.name = self.get_font_for_language(document_data['target_lang'])
            run.font.size = Pt(paragraph_data.get('font_size', 12))
            
        # Save to buffer
        buffer = BytesIO()
        doc.save(buffer)
        return buffer.getvalue()
```

### Page Range Selection Parser
```python
def parse_page_range(range_string: str, total_pages: int) -> List[int]:
    """
    Parse page range string like "1-5, 7, 10-15"
    Returns list of page numbers to include
    """
    pages = set()
    parts = range_string.split(',')
    
    for part in parts:
        part = part.strip()
        if '-' in part:
            start, end = part.split('-')
            start = max(1, int(start))
            end = min(total_pages, int(end))
            pages.update(range(start, end + 1))
        else:
            page = int(part)
            if 1 <= page <= total_pages:
                pages.add(page)
                
    return sorted(list(pages))
```

### Export Queue Management
```python
@celery_app.task(bind=True)
def generate_export_task(self, document_id: str, format: str, options: dict):
    try:
        # Update task progress
        self.update_state(state='PROCESSING', meta={'progress': 0})
        
        # Load document and translation
        document = Document.query.get(document_id)
        translation = Translation.query.filter_by(document_id=document_id).first()
        
        # Generate export based on format
        if format == 'pdf':
            exporter = PDFExporter()
            content = exporter.generate_pdf(document, translation, options)
        elif format == 'docx':
            exporter = DOCXExporter()
            content = exporter.generate_docx(document, translation, options)
        elif format == 'txt':
            content = translation.translated_text.encode('utf-8')
            
        # Upload to S3
        s3_key = f"exports/{document_id}/{format}/{uuid4()}.{format}"
        s3_client.put_object(Bucket=EXPORT_BUCKET, Key=s3_key, Body=content)
        
        # Save export record
        export = Export(
            document_id=document_id,
            format=format,
            s3_path=s3_key,
            file_size=len(content),
            status='completed'
        )
        db.session.add(export)
        db.session.commit()
        
        return {'export_id': str(export.id), 'status': 'completed'}
        
    except Exception as e:
        self.update_state(state='FAILURE', meta={'error': str(e)})
        raise
```

### Database Models
```python
class Export(Base):
    __tablename__ = 'exports'
    
    id = Column(UUID, primary_key=True)
    document_id = Column(UUID, ForeignKey('documents.id'))
    format = Column(String(10))  # pdf, docx, txt
    s3_path = Column(String(500))
    file_size = Column(Integer)
    page_range = Column(String(100))  # "1-5, 10-20"
    status = Column(String(20))  # generating, completed, failed
    created_at = Column(DateTime, default=datetime.utcnow)
    
class ShareLink(Base):
    __tablename__ = 'share_links'
    
    id = Column(UUID, primary_key=True)
    export_id = Column(UUID, ForeignKey('exports.id'))
    token = Column(String(64), unique=True)
    password_hash = Column(String(255))
    expires_at = Column(DateTime)
    download_count = Column(Integer, default=0)
    max_downloads = Column(Integer)  # Optional limit
```

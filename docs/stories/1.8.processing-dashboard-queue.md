## Story
**As a** User,
**I want** to monitor and manage all my document processing jobs,
**so that** I can track progress and prioritize important tasks.

## Acceptance Criteria
1. Dashboard showing all documents with status (uploaded, processing, completed, failed)
2. Real-time progress updates via WebSocket
3. Queue visualization with position and estimated time
4. Ability to cancel, retry, or prioritize jobs
5. Filter and search capabilities for documents
6. Processing statistics and success rates
7. Error details and retry options for failed jobs
8. Batch operations for managing multiple jobs

### WebSocket Event Structure
```typescript
interface ProcessingEvent {
  type: 'progress' | 'completed' | 'failed' | 'queued'
  documentId: string
  jobId: string
  progress?: {
    current: number
    total: number
    stage: 'uploading' | 'extracting' | 'translating' | 'exporting'
  }
  error?: {
    message: string
    code: string
    retryable: boolean
  }
  metadata?: {
    processingTime: number
    pagesProcessed: number
    tokensUsed: number
  }
}

// WebSocket connection
const ws = new WebSocket(`wss://${API_URL}/ws/processing`)

ws.onmessage = (event) => {
  const data: ProcessingEvent = JSON.parse(event.data)
  
  switch(data.type) {
    case 'progress':
      updateDocumentProgress(data.documentId, data.progress)
      break
    case 'completed':
      markDocumentComplete(data.documentId)
      showNotification('Document processed successfully')
      break
    case 'failed':
      markDocumentFailed(data.documentId, data.error)
      showErrorNotification(data.error.message)
      break
  }
}
```

from celery import Celery
from kombu import Queue, Exchange

# Celery configuration
celery_app = Celery('translation_platform')
celery_app.conf.update(
    broker_url='redis://localhost:6379/0',
    result_backend='redis://localhost:6379/0',

### Queue Management Service
```python
class QueueService:
    def get_queue_status(self) -> dict:
        """Get current queue statistics"""
        i = celery_app.control.inspect()
        
        return {
            'active': len(i.active() or {}),
            'scheduled': len(i.scheduled() or {}),
            'reserved': len(i.reserved() or {}),
            'stats': i.stats(),
            'registered': i.registered()
        }
    
    def cancel_job(self, job_id: str) -> bool:
        """Cancel a running or queued job"""
        celery_app.control.revoke(job_id, terminate=True)
        return True
    
    def prioritize_job(self, job_id: str, priority: int) -> bool:
        """Change job priority (requires requeuing)"""
        # Get task info
        result = AsyncResult(job_id)
        
        if result.state == 'PENDING':
            # Revoke old task
            celery_app.control.revoke(job_id)
            
            # Requeue with new priority
            task = celery_app.tasks[result.task_name]
            task.apply_async(
                args=result.args,
                kwargs=result.kwargs,
                priority=priority
            )
            return True
        return False
    
    def retry_failed_job(self, job_id: str) -> str:
        """Retry a failed job"""
        result = AsyncResult(job_id)
        
        if result.state == 'FAILURE':
            # Get original task info from database
            job = Job.query.get(job_id)
            
            # Create new task with same parameters
            task = celery_app.tasks[job.task_name]
            new_result = task.apply_async(
                args=job.args,
                kwargs=job.kwargs
            )
            
            return new_result.id
        return None
```

### Dashboard Statistics Queries
```python
class DashboardService:
    async def get_dashboard_stats(self, tenant_id: UUID, date_range: tuple) -> dict:
        """Get dashboard statistics for a tenant"""
        
        # Base query with tenant filter
        base_query = db.session.query(Document).filter(
            Document.tenant_id == tenant_id,
            Document.created_at.between(*date_range)
        )
        
        # Calculate statistics
        total = base_query.count()
        completed = base_query.filter(Document.status == 'completed').count()
        processing = base_query.filter(Document.status == 'processing').count()
        failed = base_query.filter(Document.status == 'failed').count()
        
        # Average processing time
        avg_time = db.session.query(
            func.avg(Document.processing_time)
        ).filter(
            Document.tenant_id == tenant_id,
            Document.status == 'completed'
        ).scalar() or 0
        
        # Success rate
        success_rate = (completed / total * 100) if total > 0 else 0
        
        return {
            'total': total,
            'completed': completed,
            'processing': processing,
            'failed': failed,
            'success_rate': round(success_rate, 2),
            'avg_processing_time': round(avg_time, 2),
            'queue_length': self.get_queue_length()
        }
```

### Database Models
```python
class Job(Base):
    __tablename__ = 'jobs'
    
    id = Column(String(50), primary_key=True)  # Celery task ID
    document_id = Column(UUID, ForeignKey('documents.id'))
    task_name = Column(String(100))
    status = Column(String(20))  # pending, processing, completed, failed
    priority = Column(Integer, default=5)
    progress = Column(Integer, default=0)
    total = Column(Integer)
    stage = Column(String(50))  # Current processing stage
    error_message = Column(Text)
    error_code = Column(String(50))
    retryable = Column(Boolean, default=True)
    created_at = Column(DateTime, default=datetime.utcnow)
    started_at = Column(DateTime)
    completed_at = Column(DateTime)
    processing_time = Column(Float)  # seconds
    args = Column(JSONB)  # Original task arguments
    kwargs = Column(JSONB)  # Original task kwargs
```
